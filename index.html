MENANGANI REQUEST GET (Data Guru, Izin Personal, Dinas Luar, & Log Presensi) // ========================================================================= function doGet(e) { if (!e || !e.parameter) { return ContentService.createTextOutput(JSON.stringify({ "status": "success", "message": "API SIGAP aktif." })).setMimeType(ContentService.MimeType.JSON); } var action = e.parameter.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); // SINKRONISASI DATA GURU (Diambil Dinamis dari Sheet "Guru") if (action === "get_users") { try { var sheet = ss.getSheetByName("Guru"); if (!sheet) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": "Sheet Guru tidak ada" })).setMimeType(ContentService.MimeType.JSON); } var rows = sheet.getDataRange().getValues(); var users = {}; for (var i = 1; i < rows.length; i++) { var nip = rows[i][0].toString().trim(); if (nip !== "") { users[nip] = { nip: nip, nama: rows[i][1] ? rows[i][1].toString().trim() : "", // Kolom 2 Nama jabatan: rows[i][2] ? rows[i][2].toString().trim() : "", // Kolom 3 Jabatan foto: rows[i][3] ? rows[i][3].toString() : "" }; } } if (e.parameter.callback) { return ContentService.createTextOutput(e.parameter.callback + "(" + JSON.stringify({ "status": "success", "users": users }) + ");").setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(JSON.stringify({ "status": "success", "users": users })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": err.toString() })).setMimeType(ContentService.MimeType.JSON); } } // SINKRONISASI STATUS DINAS LUAR (DARI SHEET "Izin DL" KOLOM 9) if (action === "get_dl") { try { var nipReq = e.parameter.nip ? e.parameter.nip.toString().trim() : ""; var sheetDL = ss.getSheetByName("Izin DL"); var dList = []; if (sheetDL) { var rowsDL = sheetDL.getDataRange().getValues(); for (var i = 1; i < rowsDL.length; i++) { var nip = rowsDL[i][1] ? rowsDL[i][1].toString().trim() : ""; if (nip === nipReq) { dList.push({ id: i, nip: nip, nama: rowsDL[i][2] ? rowsDL[i][2].toString() : "", jabatan: rowsDL[i][3] ? rowsDL[i][3].toString() : "", tipe: rowsDL[i][4] ? rowsDL[i][4].toString() : "Dinas Luar Full", tglMulai: rowsDL[i][5] ? formatDateYMD(rowsDL[i][5]) : "", tglSelesai: rowsDL[i][6] ? formatDateYMD(rowsDL[i][6]) : "", keterangan: rowsDL[i][7] ? rowsDL[i][7].toString() : "", status: rowsDL[i][8] ? rowsDL[i][8].toString().trim() : "Menunggu" }); } } } if (e.parameter.callback) { return ContentService.createTextOutput(e.parameter.callback + "(" + JSON.stringify({ "status": "success", "dinasLuar": dList }) + ");").setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(JSON.stringify({ "status": "success", "dinasLuar": dList })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": err.toString() })).setMimeType(ContentService.MimeType.JSON); } } // SINKRONISASI IZIN PERSONAL (DARI SHEET "izin_personal" KOLOM 8) if (action === "get_izin") { try { var nipReq = e.parameter.nip ? e.parameter.nip.toString().trim() : ""; var sheetIzin = ss.getSheetByName("izin_personal"); var iList = []; if (sheetIzin) { var rowsIzin = sheetIzin.getDataRange().getValues(); for (var i = 1; i < rowsIzin.length; i++) { var nip = rowsIzin[i][1] ? rowsIzin[i][1].toString().trim() : ""; if (nip === nipReq) { iList.push({ id: i, nip: nip, nama: rowsIzin[i][2] ? rowsIzin[i][2].toString() : "", jenis: rowsIzin[i][3] ? rowsIzin[i][3].toString() : "", tanggal: rowsIzin[i][4] ? formatDateYMD(rowsIzin[i][4]) : "", jam: rowsIzin[i][5] ? rowsIzin[i][5].toString() : "", keterangan: rowsIzin[i][6] ? rowsIzin[i][6].toString() : "", status: rowsIzin[i][7] ? rowsIzin[i][7].toString().trim() : "Menunggu" }); } } } if (e.parameter.callback) { return ContentService.createTextOutput(e.parameter.callback + "(" + JSON.stringify({ "status": "success", "izinPersonal": iList }) + ");").setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(JSON.stringify({ "status": "success", "izinPersonal": iList })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": err.toString() })).setMimeType(ContentService.MimeType.JSON); } } // SINKRONISASI LOG PRESENSI UNTUK LIVE KEHADIRAN (KOLOM 3, 4, 5, 6) if (action === "get_presensi") { try { var sheetPresensi = ss.getSheetByName("Presensi"); var pList = []; if (sheetPresensi) { var rowsPresensi = sheetPresensi.getDataRange().getValues(); for (var i = 1; i < rowsPresensi.length; i++) { pList.push({ nip: rowsPresensi[i][1] ? rowsPresensi[i][1].toString().trim() : "", nama: rowsPresensi[i][2] ? rowsPresensi[i][2].toString() : "", jabatan: rowsPresensi[i][3] ? rowsPresensi[i][3].toString() : "", tanggal: rowsPresensi[i][4] ? formatDateYMD(rowsPresensi[i][4]) : "", jam: rowsPresensi[i][5] ? rowsPresensi[i][5].toString() : "", status: rowsPresensi[i][6] ? rowsPresensi[i][6].toString() : "MASUK", foto: rowsPresensi[i][8] ? rowsPresensi[i][8].toString() : "" }); } } if (e.parameter.callback) { return ContentService.createTextOutput(e.parameter.callback + "(" + JSON.stringify({ "status": "success", "presensi": pList }) + ");").setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(JSON.stringify({ "status": "success", "presensi": pList })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": err.toString() })).setMimeType(ContentService.MimeType.JSON); } } } // Helper Format Tanggal YYYY-MM-DD function formatDateYMD(dateVal) { if (dateVal instanceof Date) { var y = dateVal.getFullYear(); var m = String(dateVal.getMonth() + 1).padStart(2, '0'); var d = String(dateVal.getDate()).padStart(2, '0'); return y + "-" + m + "-" + d; } return dateVal ? dateVal.toString() : ""; } // ========================================================================= // 2. MENANGANI REQUEST POST (Menulis Pengajuan & Absen Baru) // ========================================================================= function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": "POST data kosong." })).setMimeType(ContentService.MimeType.JSON); } var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); // VALIDASI UTAMA: Cari Nama & Jabatan Real-Time dari Sheet Guru berdasarkan NIP input var sheetGuru = ss.getSheetByName("Guru"); var namaAsli = data.nama; // Fallback default bawaan kiriman aplikasi var jabatanAsli = data.jabatan; // Fallback default bawaan kiriman aplikasi if (sheetGuru && data.nip) { var rowsGuru = sheetGuru.getDataRange().getValues(); var targetNip = data.nip.toString().trim(); for (var g = 1; g < rowsGuru.length; g++) { var nipDiSheet = rowsGuru[g][0].toString().trim(); if (nipDiSheet === targetNip) { namaAsli = rowsGuru[g][1] ? rowsGuru[g][1].toString().trim() : namaAsli; jabatanAsli = rowsGuru[g][2] ? rowsGuru[g][2].toString().trim() : jabatanAsli; break; } } } // MENYIMPAN AJUAN IZIN PERSONAL if (data.action === "add_izin") { var sheetIzin = ss.getSheetByName("izin_personal"); if (!sheetIzin) { sheetIzin = ss.insertSheet("izin_personal"); sheetIzin.appendRow(["Timestamp", "NIP", "Nama", "Jenis Izin", "Tanggal", "Estimasi Jam", "Keterangan", "Status"]); } sheetIzin.appendRow([ new Date(), data.nip.toString(), namaAsli, // Data Akurat Kolom 2 data.jenis, data.tanggal, data.jam, data.keterangan, "Menunggu" ]); return ContentService.createTextOutput(JSON.stringify({ "status": "success", "message": "Izin Berhasil Diajukan" })).setMimeType(ContentService.MimeType.JSON); } // MENYIMPAN AJUAN DL BARU if (data.action === "add_dl") { var sheetDL = ss.getSheetByName("Izin DL"); if (!sheetDL) { sheetDL = ss.insertSheet("Izin DL"); sheetDL.appendRow(["Timestamp", "NIP", "Nama", "Jabatan", "Tipe Dinas Luar", "Tanggal Mulai", "Tanggal Selesai", "Keterangan", "Status"]); } sheetDL.appendRow([ new Date(), data.nip.toString(), namaAsli, // Data Akurat Kolom 2 jabatanAsli, // Data Akurat Kolom 3 data.tipe, data.tglMulai, data.tglSelesai, data.keterangan, "Menunggu" ]); return ContentService.createTextOutput(JSON.stringify({ "status": "success", "message": "DL Berhasil Diajukan" })).setMimeType(ContentService.MimeType.JSON); } // ABSEN REGULER var sheetPresensi = ss.getSheetByName("Presensi"); if (!sheetPresensi) { sheetPresensi = ss.insertSheet("Presensi"); sheetPresensi.appendRow(["Timestamp", "NIP", "Nama", "Jabatan", "Tanggal", "Jam", "Tipe Absen", "Koordinat GPS", "Foto"]); } sheetPresensi.appendRow([ new Date(), data.nip.toString(), namaAsli, // Otomatis tersimpan Nama asli Kolom 2 Spreadsheet jabatanAsli, // Otomatis tersimpan Jabatan asli Kolom 3 Spreadsheet data.tanggal, data.jam, data.tipe, data.koordinat, data.foto ]); return ContentService.createTextOutput(JSON.stringify({ "status": "success" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
